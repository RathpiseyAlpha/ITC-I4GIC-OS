Copy, rename and delete

student@server: ~/oslab-work/lab1/task2_files

$ cp FILE1.txt FILE1_copy.txt

$ ls >> ../task2_file_commands.txt

$ mv FILE2.txt FILE2_renamed.txt

$ ls >> ../task2_file_commands.txt

$ rm FILE1_copy.txt

$ ls >> ../task2_file_commands.txt

$ cd ..

$ cat task2_file_commands.txt
